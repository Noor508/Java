import javax.servlet.ServletException;

import javax.servlet.http.HttpServletResponse;

public class BookServlet extends HttpServlet {
    private ArrayList<Book> bookList;

    @Override
    public void init() throws ServletException {
        bookList = new ArrayList<>();
        bookList.add(new Book("Java how to program", "Deitel", "9780136053064"));
        bookList.add(new Book("Head First C#", "Andrew Stellman", "9781491976708"));
        bookList.add(new Book("C#", "Svetlin Nakov", "9789544007737"));
    }

    @Override
    protected void doGet(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        response.setContentType("text/html");
        PrintWriter out = response.getWriter();

        out.println("<html><head><title>Book Collection</title></head><body>");
        out.println("<h1>Book Collection</h1>");
        out.println("<table border='1'>");
        out.println("<tr><th>Title</th><th>Author</th><th>ISBN</th></tr>");

        for (Book book : bookList) {
            out.println("<tr>");
            out.println("<td>" + book.getTitle() + "</td>");
            out.println("<td>" + book.getAuthor() + "</td>");
            out.println("<td>" + book.getISBN() + "</td>");
            out.println("</tr>");
        }

        out.println("</table>");
        out.println("<h1>Add Book</h1>");
        out.println("<form action='/books' method='post'>");
        out.println("<input type='hidden' name='action' value='create'>");
        out.println("<label>Title:</label>");
        out.println("<input type='text' name='title'><br>");
        out.println("<label>Author:</label>");
        out.println("<input type='text' name='author'><br>");
        out.println("<label>ISBN:</label>");
        out.println("<input type='text' name='isbn'><br>");
        out.println("<input type='submit' value='Add Book'>");
        out.println("</form>");

        out.println("<h1>Update Book</h1>");
        out.println("<form action='/books' method='post'>");
        out.println("<input type='hidden' name='action' value='update'>");
        out.println("<label>ISBN:</label>");
        out.println("<input type='text' name='isbn'><br>");
        out.println("<label>New Title:</label>");
        out.println("<input type='text' name='title'><br>");
        out.println("<label>New Author:</label>");
        out.println("<input type='text' name='author'><br>");
        out.println("<input type='submit' value='Update Book'>");
        out.println("</form>");

        out.println("<h1>Delete Book</h1>");
        out.println("<form action='/books' method='post'>");
        out.println("<input type='hidden' name='action' value='delete'>");
        out.println("<label>ISBN:</label>");
        out.println("<input type='text' name='isbn'><br>");
        out.println("<input type='submit' value='Delete Book'>");
        out.println("</form>");

        out.println("</body></html>");
    }

    @Override
    protected void doPost(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        String action = request.getParameter("action");

        if (action != null) {
            switch (action) {
                case "create":
                    String title = request.getParameter("title");
                    String author = request.getParameter("author");
                    String isbn = request.getParameter("isbn");
                    bookList.add(new Book(title, author, isbn));
                    break;
                case "update":
                    isbn = request.getParameter("isbn");
                    for (Book book : bookList) {
                        if (book.getISBN().equals(isbn)) {
                            title = request.getParameter("title");
                            author = request.getParameter("author");
                            book.setTitle(title);
                            book.setAuthor(author);
                            break;
                        }
                    }
                    break;
                case "delete":
                    isbn = request.getParameter("isbn");
                    bookList.removeIf(book -> book.getISBN().equals(isbn));
                    break;
            }
        }

        doGet(request, response);
    }
}
